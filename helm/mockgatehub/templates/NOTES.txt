  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "mockgatehub.fullname" . }}-admin {{ .Values.adminService.port }}:{{ .Values.adminService.port }}
  open http://localhost:{{ .Values.adminService.port }}/ui

{{ end -}}
